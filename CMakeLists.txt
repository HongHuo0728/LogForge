cmake_minimum_required(VERSION 3.24)
project(LogForge VERSION 0.1.0 LANGUAGES CXX)
enable_language(RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "LogForge v1 requires Windows x64.")
endif()
if(NOT MSVC OR NOT MSVC_CXX_ARCHITECTURE_ID STREQUAL "x64")
  message(FATAL_ERROR "The v1 Windows backend currently requires MSVC targeting x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
add_library(logforge_core src/color/AppleLog.cpp src/color/HLG.cpp src/Platform.cpp src/Media.cpp src/FFmpeg.cpp src/Transcode.cpp)
target_include_directories(logforge_core PUBLIC include)
target_include_directories(logforge_core SYSTEM PUBLIC third_party)
target_link_libraries(logforge_core PUBLIC shell32 ole32 bcrypt winhttp)
add_executable(LogForge-cli src/Cli.cpp)
target_link_libraries(LogForge-cli PRIVATE logforge_core)
add_executable(LogForge WIN32 src/MainWindow.cpp resources/LogForge.rc)
target_link_libraries(LogForge PRIVATE logforge_core comctl32 dwmapi windowscodecs uxtheme)
target_include_directories(LogForge PRIVATE resources)
target_link_options(LogForge PRIVATE /MANIFEST:NO)
enable_testing()
add_executable(logforge_tests tests/tests.cpp)
target_link_libraries(logforge_tests PRIVATE logforge_core)
add_test(NAME color_math COMMAND logforge_tests color)
add_test(NAME media_metadata_validation COMMAND logforge_tests media)
add_test(NAME ffmpeg_platform COMMAND logforge_tests platform)
set_tests_properties(color_math media_metadata_validation ffmpeg_platform PROPERTIES ENVIRONMENT "LOGFORGE_DATA_DIR=${CMAKE_BINARY_DIR}/test-data")
set(LOGFORGE_TEST_FFMPEG "" CACHE FILEPATH "ffmpeg.exe for optional end-to-end integration tests")
if(LOGFORGE_TEST_FFMPEG)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME media_pipeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py --cli $<TARGET_FILE:LogForge-cli> --ffmpeg ${LOGFORGE_TEST_FFMPEG} --work ${CMAKE_BINARY_DIR}/integration)
  set_tests_properties(media_pipeline PROPERTIES TIMEOUT 180)
endif()
install(TARGETS LogForge RUNTIME DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-MIT.txt)
install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md" PATTERN "*.png")
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME LogForge)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "LogForge-${PROJECT_VERSION}-Windows-x64")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
include(CPack)
