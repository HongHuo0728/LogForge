cmake_minimum_required(VERSION 3.24)
project(LogForge VERSION 1.0.0 LANGUAGES CXX)
set(LOGFORGE_BUILD_NUMBER "26922A")
configure_file(resources/Version.h.in generated/logforge/Version.h @ONLY)
configure_file(resources/VersionResource.h.in generated/VersionResource.h @ONLY)
configure_file(resources/LogForge.manifest generated/LogForge.manifest @ONLY)
enable_language(RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "LogForge v1 requires Windows x64.")
endif()
if(NOT MSVC OR NOT MSVC_CXX_ARCHITECTURE_ID STREQUAL "x64")
  message(FATAL_ERROR "The v1 Windows backend currently requires MSVC targeting x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
add_library(logforge_core src/color/AppleLog.cpp src/color/HLG.cpp src/Platform.cpp src/Media.cpp src/FFmpeg.cpp src/Transcode.cpp src/Localization.cpp src/Settings.cpp src/Discovery.cpp)
target_include_directories(logforge_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(logforge_core SYSTEM PUBLIC third_party)
target_link_libraries(logforge_core PUBLIC shell32 ole32 bcrypt winhttp)
add_executable(LogForge-cli src/Cli.cpp resources/LogForge.rc)
target_compile_definitions(LogForge-cli PRIVATE LOGFORGE_CLI)
target_include_directories(LogForge-cli PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_options(LogForge-cli PRIVATE /MANIFEST:NO)
target_link_libraries(LogForge-cli PRIVATE logforge_core)
add_executable(LogForge WIN32 src/MainWindow.cpp src/Ui.cpp src/SettingsWindow.cpp resources/LogForge.rc)
target_link_libraries(LogForge PRIVATE logforge_core comctl32 dwmapi windowscodecs uxtheme)
target_include_directories(LogForge PRIVATE resources "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_options(LogForge PRIVATE /MANIFEST:NO)
enable_testing()
add_executable(logforge_tests tests/tests.cpp)
target_link_libraries(logforge_tests PRIVATE logforge_core)
add_test(NAME color_math COMMAND logforge_tests color)
add_test(NAME media_metadata_validation COMMAND logforge_tests media)
add_test(NAME ffmpeg_platform COMMAND logforge_tests platform)
set_tests_properties(color_math media_metadata_validation ffmpeg_platform PROPERTIES ENVIRONMENT "LOGFORGE_DATA_DIR=${CMAKE_BINARY_DIR}/test-data")
add_executable(logforge_application_tests tests/application.cpp)
target_link_libraries(logforge_application_tests PRIVATE logforge_core version)
add_dependencies(logforge_application_tests LogForge LogForge-cli)
add_test(NAME application_preferences COMMAND logforge_application_tests preferences)
add_test(NAME ffmpeg_disk_discovery COMMAND logforge_application_tests discovery)
add_test(NAME application_resources COMMAND logforge_application_tests resources)
set_tests_properties(application_preferences ffmpeg_disk_discovery application_resources PROPERTIES
  ENVIRONMENT "LOGFORGE_DATA_DIR=${CMAKE_BINARY_DIR}/test-data")
set(LOGFORGE_TEST_FFMPEG "" CACHE FILEPATH "ffmpeg.exe for optional end-to-end integration tests")
if(LOGFORGE_TEST_FFMPEG)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME media_pipeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py --cli $<TARGET_FILE:LogForge-cli> --ffmpeg ${LOGFORGE_TEST_FFMPEG} --work ${CMAKE_BINARY_DIR}/integration)
  set_tests_properties(media_pipeline PROPERTIES TIMEOUT 180)
  add_test(NAME color_signal COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/signal_integrity.py --cli $<TARGET_FILE:LogForge-cli> --ffmpeg ${LOGFORGE_TEST_FFMPEG} --work ${CMAKE_BINARY_DIR}/signal-test)
  set_tests_properties(color_signal PROPERTIES TIMEOUT 90)
  add_test(NAME ffmpeg_discovery_capabilities COMMAND logforge_application_tests discovery ${LOGFORGE_TEST_FFMPEG})
  set_tests_properties(ffmpeg_discovery_capabilities PROPERTIES TIMEOUT 90
    ENVIRONMENT "LOGFORGE_DATA_DIR=${CMAKE_BINARY_DIR}/test-data")
endif()
install(TARGETS LogForge RUNTIME DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-MIT.txt)
install(FILES docs/ARCHITECTURE.md docs/COLOR_PIPELINE.md docs/CREATIVE_ADJUSTMENTS.md
  docs/FFMPEG_PROVIDER.md docs/ICON.md docs/METADATA.md docs/TECHNICAL_REFERENCES.md
  docs/VALIDATION.md DESTINATION docs)
install(FILES docs/images/LogForge.png DESTINATION docs/images)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME LogForge)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "LogForge-${PROJECT_VERSION}-Windows-x64")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
include(CPack)
